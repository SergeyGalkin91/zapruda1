type Competition = {
  id: string;
  organizer_id: string;
  title: string;
  description: string;
  area_label: string;
  starts_at: string;
  ends_at: string;
  status: "draft" | "open" | "live" | "finished" | "cancelled";
  max_teams: number;
  scoring_mode: "total_fish" | "total_weight_g" | "biggest_fish_g";
};

type Team = {
  id: string;
  competition_id: string;
  club_id: string | null;
  captain_id: string;
  name: string;
  status: "registered" | "withdrawn" | "disqualified";
};

type Result = {
  id: string;
  competition_id: string;
  team_id: string;
  submitted_by: string;
  fish_count: number;
  total_weight_g: number;
  biggest_fish_g: number;
  note: string;
  status: "pending" | "accepted" | "rejected";
};

  biggest_fish_g: "По самой крупной рыбе",
};

function scoreValue(mode: Competition["scoring_mode"], r: Result) {
  if (mode === "total_weight_g") return r.total_weight_g;
  if (mode === "biggest_fish_g") return r.biggest_fish_g;
  return r.fish_count;
}
